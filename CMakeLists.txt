# --- The Rust FFI library ----------------------------------------------------
#
# Either way the result is a `kameleoon_ffi` target that kameleoon_cpp_client
# links PUBLIC, so consumers only ever name Kameleoon::CppClient.

if(KAMELEOON_FFI_LIBRARY)
    # Prebuilt shared library (cdylib produced by `cargo build -p kameleoon-ffi`).
    set(_kameleoon_ffi_is_shared ON)
    add_library(kameleoon_ffi SHARED IMPORTED GLOBAL)
    set_target_properties(kameleoon_ffi PROPERTIES IMPORTED_LOCATION "${KAMELEOON_FFI_LIBRARY}")

    if(WIN32)
        # MSVC-style linking needs the import library produced by rustc next to
        # the DLL (kameleoon_ffi.dll.lib); MinGW toolchains can produce
        # libkameleoon_ffi.dll.a instead.
        if(NOT KAMELEOON_FFI_IMPLIB)
            get_filename_component(_kameleoon_ffi_dir "${KAMELEOON_FFI_LIBRARY}" DIRECTORY)
            find_file(_kameleoon_ffi_found_implib
                NAMES kameleoon_ffi.dll.lib kameleoon_ffi.lib libkameleoon_ffi.dll.a
                PATHS "${_kameleoon_ffi_dir}"
                NO_DEFAULT_PATH
            )
            if(_kameleoon_ffi_found_implib)
                set(KAMELEOON_FFI_IMPLIB "${_kameleoon_ffi_found_implib}")
            endif()
        endif()
        if(NOT KAMELEOON_FFI_IMPLIB)
            message(FATAL_ERROR
                "Found ${KAMELEOON_FFI_LIBRARY} but no import library (kameleoon_ffi.dll.lib). "
                "Pass -DKAMELEOON_FFI_IMPLIB=/path/to/kameleoon_ffi.dll.lib.")
        endif()
        set_target_properties(kameleoon_ffi PROPERTIES IMPORTED_IMPLIB "${KAMELEOON_FFI_IMPLIB}")
    endif()
else()
    # Build from source: cargo compiles native/kameleoon_ffi as a static
    # library that is linked into the consumer's binary, so nothing has to be
    # deployed next to it. Corrosion drives cargo from CMake and adds the
    # system libraries the Rust standard library needs (it asks rustc, so the
    # list never has to be maintained here).
    set(_kameleoon_ffi_is_shared OFF)

    find_program(KAMELEOON_CARGO cargo
        HINTS "$ENV{CARGO_HOME}/bin" "$ENV{HOME}/.cargo/bin" "$ENV{USERPROFILE}/.cargo/bin")
    if(NOT KAMELEOON_CARGO)
        message(FATAL_ERROR
            "cargo not found: building the Kameleoon FFI from source needs a Rust toolchain "
            "(https://rustup.rs). Alternatively pass -DKAMELEOON_FFI_LIBRARY=/path/to/libkameleoon_ffi "
            "and -DKAMELEOON_FFI_HEADER=/path/to/sdk-core.h to use a prebuilt library.")
    endif()

    # Corrosion: already provided by the consumer, installed on the system,
    # vendored under cmake/corrosion/, or fetched (pinned) from GitHub.
    if(NOT COMMAND corrosion_import_crate)
        find_package(Corrosion 0.5 QUIET CONFIG)
        if(NOT Corrosion_FOUND)
            if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/corrosion/CMakeLists.txt")
                add_subdirectory(cmake/corrosion)
            else()
                include(FetchContent)
                FetchContent_Declare(Corrosion
                    GIT_REPOSITORY https://github.com/corrosion-rs/corrosion.git
                    GIT_TAG v0.6.1
                    GIT_SHALLOW TRUE
                )
                FetchContent_MakeAvailable(Corrosion)
            endif()
        endif()
    endif()

    # The published package ships Cargo.lock; honour it. The monorepo manifest
    # has none (it is not a workspace member), so cargo resolves it there.
    get_filename_component(_kameleoon_ffi_manifest_dir "${KAMELEOON_FFI_MANIFEST}" DIRECTORY)
    set(_kameleoon_ffi_locked "")
    if(EXISTS "${_kameleoon_ffi_manifest_dir}/Cargo.lock")
        set(_kameleoon_ffi_locked LOCKED)
    endif()

    # Creates the `kameleoon_ffi` target (and `kameleoon_ffi-static` behind it).
    corrosion_import_crate(
        MANIFEST_PATH "${KAMELEOON_FFI_MANIFEST}"
        CRATES kameleoon-ffi
        PROFILE "${KAMELEOON_FFI_PROFILE}"
        ${_kameleoon_ffi_locked}
    )
endif()

target_link_libraries(kameleoon_cpp_client PUBLIC kameleoon_ffi Threads::Threads)

# --- The C header of the Rust FFI ------------------------------------------
#
# The C++ sources include `sdk-core.h`, generated by cbindgen from the Rust
# sources, so the ABI is never mirrored by hand. The published package bundles
# the header generated for the crate sources it ships; otherwise point
# KAMELEOON_FFI_HEADER at the header matching a prebuilt FFI library, or let
# the monorepo build generate it (requires `cargo install cbindgen --locked`).

set(_kameleoon_ffi_header_dir "${CMAKE_CURRENT_BINARY_DIR}/ffi-header")
file(MAKE_DIRECTORY "${_kameleoon_ffi_header_dir}")

if(NOT KAMELEOON_FFI_HEADER AND EXISTS "${_kameleoon_bundled_header}")
    set(KAMELEOON_FFI_HEADER "${_kameleoon_bundled_header}")
endif()

if(KAMELEOON_FFI_HEADER)
    configure_file("${KAMELEOON_FFI_HEADER}" "${_kameleoon_ffi_header_dir}/sdk-core.h" COPYONLY)
else()
    find_program(KAMELEOON_CBINDGEN cbindgen)
    if(NOT KAMELEOON_CBINDGEN)
        message(FATAL_ERROR
            "cbindgen not found. Install it with `cargo install cbindgen --locked`, or pass "
            "-DKAMELEOON_FFI_HEADER=/path/to/sdk-core.h generated for the FFI library in use.")
    endif()
    file(GLOB_RECURSE KAMELEOON_FFI_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/../../core/ffi/src/*.rs")
    add_custom_command(
        OUTPUT "${_kameleoon_ffi_header_dir}/sdk-core.h"
        COMMAND "${KAMELEOON_CBINDGEN}" --quiet
            --config "${CMAKE_CURRENT_SOURCE_DIR}/../../cbindgen.toml"
            --crate kameleoon-ffi --output "${_kameleoon_ffi_header_dir}/sdk-core.h"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.."
        DEPENDS ${KAMELEOON_FFI_SOURCES} ../../cbindgen.toml
            ../../core/ffi/Cargo.toml ../../Cargo.toml ../../Cargo.lock
        COMMENT "Generating sdk-core.h from core/ffi with cbindgen"
        VERBATIM
    )
    add_custom_target(kameleoon_ffi_header DEPENDS "${_kameleoon_ffi_header_dir}/sdk-core.h")
endif()

# Every target that compiles anything from src/ (directly or through a header
# in src/detail/) needs the generated header, and must wait for cbindgen.
# The header is an implementation detail: consumers only see include/.
function(kameleoon_use_ffi_header target)
    target_include_directories(${target} PRIVATE "${_kameleoon_ffi_header_dir}")
    if(TARGET kameleoon_ffi_header)
        add_dependencies(${target} kameleoon_ffi_header)
    endif()
endfunction()

kameleoon_use_ffi_header(kameleoon_cpp_client)

# --- Runtime lookup helpers for executables ----------------------------------

if(_kameleoon_ffi_is_shared)
    get_filename_component(KAMELEOON_FFI_DIRECTORY "${KAMELEOON_FFI_LIBRARY}" DIRECTORY)
endif()

# Only needed with a prebuilt shared FFI: the static one is linked in. RPATH
# only affects executables/shared libraries, so it is applied to the targets
# built here (not to the static kameleoon_cpp_client library). Consumers of
# the shared variant should apply an equivalent setting to their own
# executables (or install libkameleoon_ffi to a standard location).
function(kameleoon_apply_runtime_lookup target)
    if(NOT _kameleoon_ffi_is_shared)
        return()
    endif()
    if(APPLE)
        set_target_properties(${target} PROPERTIES
            BUILD_RPATH "@loader_path;${KAMELEOON_FFI_DIRECTORY}"
            INSTALL_RPATH "@loader_path;${KAMELEOON_FFI_DIRECTORY}"
        )
    elseif(UNIX)
        set_target_properties(${target} PROPERTIES
            BUILD_RPATH "$ORIGIN;${KAMELEOON_FFI_DIRECTORY}"
            INSTALL_RPATH "$ORIGIN;${KAMELEOON_FFI_DIRECTORY}"
        )
    elseif(WIN32)
        # Windows has no RPATH; keep the DLL next to the executable instead.
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_PROPERTY:kameleoon_ffi,IMPORTED_LOCATION>"
                "$<TARGET_FILE_DIR:${target}>"
        )
    endif()
endfunction()

if(KAMELEOON_CPP_BUILD_TESTS)
    enable_testing()

    # GoogleTest runs every test binary; cpr (with its own libcurl on the
    # platform TLS backend) lets the live client tests seed and read back
    # Kameleoon data over HTTPS. Both are fetched at configure time, so no
    # system package is needed on Windows or macOS; Linux needs OpenSSL
    # development files for libcurl.
    include(FetchContent)
    FetchContent_Declare(googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG v1.17.0
        GIT_SHALLOW TRUE
    )
    FetchContent_Declare(cpr
        GIT_REPOSITORY https://github.com/libcpr/cpr.git
        GIT_TAG 1.14.2
        GIT_SHALLOW TRUE
    )
    # Match the SDK's CRT on MSVC; otherwise gtest links against a different runtime.
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(CPR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    # cpr 1.14 builds libpsl (cookie public-suffix checks) with meson unless
    # told otherwise; the tests never set cookies, so skip that dependency.
    set(CPR_CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest cpr)
    include(GoogleTest)

    # Registers every TEST() of `target` with CTest by listing them from the
    # built binary, so a new test needs no CMake change. The prefix keeps a
    # whole binary selectable with `ctest -R/-E <target>`.
    function(kameleoon_add_gtest target)
        target_link_libraries(${target} PRIVATE GTest::gtest_main)
        gtest_discover_tests(${target} TEST_PREFIX "${target}." PROPERTIES TIMEOUT 30)
    endfunction()

    # Loads the real FFI library without credentials or network.
    add_executable(kameleoon_cpp_smoke tests/smoke.cpp)
    target_link_libraries(kameleoon_cpp_smoke PRIVATE Kameleoon::CppClient)
    kameleoon_apply_runtime_lookup(kameleoon_cpp_smoke)
    kameleoon_add_gtest(kameleoon_cpp_smoke)

    add_executable(kameleoon_cpp_header_import_test tests/header_import_test.cpp)
    target_link_libraries(kameleoon_cpp_header_import_test PRIVATE Kameleoon::CppClient)
    kameleoon_apply_runtime_lookup(kameleoon_cpp_header_import_test)
    kameleoon_add_gtest(kameleoon_cpp_header_import_test)

    # Unit tests of src/detail helpers; no FFI call is made.
    add_executable(kameleoon_cpp_detail_test tests/detail_test.cpp)
    target_link_libraries(kameleoon_cpp_detail_test PRIVATE Kameleoon::CppClient)
    kameleoon_use_ffi_header(kameleoon_cpp_detail_test)
    kameleoon_apply_runtime_lookup(kameleoon_cpp_detail_test)
    kameleoon_add_gtest(kameleoon_cpp_detail_test)

    # Exercise the public wrapper against a controlled native boundary. These
    # tests run without credentials, network access, or a background runtime.
    add_executable(kameleoon_cpp_wrapper_test tests/wrapper_test.cpp ${KAMELEOON_CPP_SOURCES})
    target_include_directories(kameleoon_cpp_wrapper_test PRIVATE include src)
    target_link_libraries(kameleoon_cpp_wrapper_test PRIVATE Threads::Threads)
    kameleoon_use_ffi_header(kameleoon_cpp_wrapper_test)
    kameleoon_add_gtest(kameleoon_cpp_wrapper_test)

    # The same suite compiled as C++20 also runs the coroutine (`co_*`) tests;
    # the library stays C++17, the awaitables are header-only.
    if("cxx_std_20" IN_LIST CMAKE_CXX_COMPILE_FEATURES)
        add_executable(kameleoon_cpp_wrapper_test_cxx20 tests/wrapper_test.cpp ${KAMELEOON_CPP_SOURCES})
        target_include_directories(kameleoon_cpp_wrapper_test_cxx20 PRIVATE include src)
        target_link_libraries(kameleoon_cpp_wrapper_test_cxx20 PRIVATE Threads::Threads)
        target_compile_features(kameleoon_cpp_wrapper_test_cxx20 PRIVATE cxx_std_20)
        kameleoon_use_ffi_header(kameleoon_cpp_wrapper_test_cxx20)
        kameleoon_add_gtest(kameleoon_cpp_wrapper_test_cxx20)
    endif()

    # Live tests against real Kameleoon services; credentials come from the
    # environment, as in the Rust and Elixir suites.
    add_executable(kameleoon_cpp_client_test tests/client_test.cpp)
    target_link_libraries(kameleoon_cpp_client_test PRIVATE Kameleoon::CppClient cpr::cpr)
    kameleoon_apply_runtime_lookup(kameleoon_cpp_client_test)
    kameleoon_add_gtest(kameleoon_cpp_client_test)

if(KAMELEOON_CPP_BUILD_EXAMPLES)
    # The example server runs on Drogon, which sits in the same throughput
    # bracket as the Rust example's actix-web, so /kam compares the SDKs
    # rather than the HTTP stacks. Drogon and its jsoncpp dependency are
    # fetched at configure time (git, network on the first build); the
    # system must provide zlib and, on Linux, libuuid (`uuid-dev`).
    include(FetchContent)
    FetchContent_Declare(jsoncpp
        GIT_REPOSITORY https://github.com/open-source-parsers/jsoncpp.git
        GIT_TAG 1.9.6
        GIT_SHALLOW TRUE
    )
    FetchContent_Declare(drogon
        GIT_REPOSITORY https://github.com/drogonframework/drogon.git
        GIT_TAG v1.9.13
        GIT_SHALLOW TRUE
    )

    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
    set(BUILD_OBJECT_LIBS OFF CACHE BOOL "" FORCE)
    set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
    set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
    set(JSONCPP_WITH_PKGCONFIG_SUPPORT OFF CACHE BOOL "" FORCE)
    set(JSONCPP_WITH_CMAKE_PACKAGE OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(jsoncpp)

    # Drogon locates jsoncpp with its own find module, which only searches
    # installed copies. Pre-seeding the module's cache entries with the
    # fetched sources and target makes it accept this in-tree build instead.
    set(JSONCPP_INCLUDE_DIRS "${jsoncpp_SOURCE_DIR}/include" CACHE PATH "" FORCE)
    set(JSONCPP_LIBRARIES jsoncpp_static CACHE STRING "" FORCE)

    # HTTP only: no ORM, drogon_ctl, examples, TLS, DNS resolver or compression.
    set(USE_SUBMODULE ON CACHE BOOL "" FORCE)
    set(BUILD_CTL OFF CACHE BOOL "" FORCE)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_ORM OFF CACHE BOOL "" FORCE)
    set(BUILD_BROTLI OFF CACHE BOOL "" FORCE)
    set(BUILD_YAML_CONFIG OFF CACHE BOOL "" FORCE)
    set(BUILD_DOC OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(BUILD_C-ARES OFF CACHE BOOL "" FORCE)
    set(TRANTOR_USE_TLS none CACHE STRING "" FORCE)
    message(STATUS "Preparing Drogon v1.9.13 and jsoncpp 1.9.6 (cached after the first build)")
    FetchContent_MakeAvailable(drogon)

    add_executable(kameleoon_cpp_server_example ../examples/server.cpp)
    target_link_libraries(kameleoon_cpp_server_example
        PRIVATE
            Kameleoon::CppClient
            Threads::Threads
            drogon
    )
    kameleoon_apply_runtime_lookup(kameleoon_cpp_server_example)
endif()
